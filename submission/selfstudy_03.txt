"""Настройки Золотой лихорадки."""
CELL = 40
MAP_ROWS = [
    "#################",
    "#...............#",
    "#.###...#...###.#",
    "#.#...........#.#",
    "#.#.###.#####.#.#",
    "#...#.......#...#",
    "#.###.#####.###.#",
    "#...............#",
    "#.###.#####.###.#",
    "#...............#",
    "#################",
]
MAP_W = len(MAP_ROWS[0])
MAP_H = len(MAP_ROWS)
PLAYER_SPEED = 160.0
PLAYER_RADIUS = 14
BOOST_SPEED = 260.0
BOOST_TIME = 1.5
COIN_RADIUS = 10
COIN_VALUE = 1
GOLD_VALUE = 3
GOLD_CHANCE = 0.25
GOLD_LIFETIME = 5.0
COINS_ON_FIELD = 5
SPAWN_INTERVAL = 1.0
MATCH_TIME = 60.0
TABLE_SIZE = 10
SPAWNS = [(1, 1), (15, 9)]
NAMES = ["Зелёный", "Жёлтый"]

# === entities.py ===
"""Старатели и монеты."""
import settings as s


class Entity:
    """Объект с координатами и радиусом."""

    def __init__(self, x: float, y: float, radius: float):
        self.x = float(x)
        self.y = float(y)
        self.radius = float(radius)
        self.alive = True

    @property
    def cell(self) -> tuple[int, int]:
        """Клетка объекта."""
        return int(self.x // s.CELL), int(self.y // s.CELL)

    def __repr__(self) -> str:
        return f"{self.__class__.__name__}(x={self.x:g}, y={self.y:g})"

    def distance_to(self, other: "Entity") -> float:
        """Расстояние между центрами."""
        return ((self.x - other.x) ** 2 + (self.y - other.y) ** 2) ** 0.5

    def overlaps(self, other: "Entity") -> bool:
        """Проверяет пересечение объектов."""
        return self.distance_to(other) < self.radius + other.radius


class Player(Entity):
    """Старатель, собирающий монеты."""

    def __init__(self, pid: int, name: str):
        cell_x, cell_y = s.SPAWNS[pid]
        super().__init__(cell_x * s.CELL + s.CELL / 2,
                         cell_y * s.CELL + s.CELL / 2, s.PLAYER_RADIUS)
        self.pid = pid
        self.name = name
        self.score = 0
        self.coins = 0
        self.boost = 0.0
        self.dir_x, self.dir_y = 1.0, 0.0
        self.input = {"dx": 0.0, "dy": 0.0}

    @property
    def speed(self) -> float:
        """Текущая скорость с учётом ускорения."""
        return s.BOOST_SPEED if self.boost > 0 else s.PLAYER_SPEED

    def take_coin(self, value: int) -> None:
        """Добавляет очки и ускоряет старателя."""
        self.score += value
        self.coins += 1
        self.boost = s.BOOST_TIME

    def to_dict(self) -> dict:
        """Снимок старателя."""
        return {"id": self.pid, "name": self.name,
                "x": round(self.x, 1), "y": round(self.y, 1),
                "dx": self.dir_x, "dy": self.dir_y,
                "score": self.score, "coins": self.coins,
                "boost": round(self.boost, 2)}


class Coin(Entity):
    """Обычная или золотая монета."""

    def __init__(self, cell_x: int, cell_y: int, gold: bool = False):
        super().__init__(cell_x * s.CELL + s.CELL / 2,
                         cell_y * s.CELL + s.CELL / 2, s.COIN_RADIUS)
        self.gold = gold
        self.value = s.GOLD_VALUE if gold else s.COIN_VALUE
        self.life = s.GOLD_LIFETIME if gold else None

    def update(self, dt: float) -> None:
        """Уменьшает время жизни золотой монеты."""
        if self.life is None:
            return
        self.life -= dt
        if self.life <= 0:
            self.life = 0.0
            self.alive = False

    def to_dict(self) -> dict:
        """Снимок монеты."""
        return {"x": round(self.x, 1), "y": round(self.y, 1),
                "gold": self.gold, "value": self.value,
                "life": None if self.life is None else round(self.life, 2)}

# === game.py ===
"""Игровое ядро Золотой лихорадки."""
import random
import settings as s
from entities import Coin, Player


class Game:
    """Правила и состояние шахты."""

    def __init__(self, player_names: list[str] | None = None):
        names = player_names or s.NAMES
        self.field = [list(row) for row in s.MAP_ROWS]
        self.free_cells = self._collect_free_cells()
        self.players = [Player(i, names[i]) for i in range(len(s.SPAWNS))]
        self.coins: list[Coin] = []
        self.time_left = s.MATCH_TIME
        self.spawn_timer = s.SPAWN_INTERVAL
        self.over = False
        self.winner_id: int | None = None
        for _ in range(s.COINS_ON_FIELD):
            self.spawn_coin()

    def _collect_free_cells(self) -> list[tuple[int, int]]:
        """Находит клетки без породы."""
        cells = []
        for y in range(s.MAP_H):
            for x in range(s.MAP_W):
                if not self.is_wall(x, y):
                    cells.append((x, y))
        return cells

    def is_wall(self, cell_x: int, cell_y: int) -> bool:
        """Проверяет породу и границу карты."""
        if not (0 <= cell_x < s.MAP_W and 0 <= cell_y < s.MAP_H):
            return True
        return self.field[cell_y][cell_x] == "#"

    def hits_wall(self, x: float, y: float, radius: float) -> bool:
        """Проверяет четыре угла вокруг объекта."""
        for corner_x in (x - radius, x + radius):
            for corner_y in (y - radius, y + radius):
                if self.is_wall(int(corner_x // s.CELL), int(corner_y // s.CELL)):
                    return True
        return False

    def spawn_coin(self) -> None:
        """Кладёт монету в случайную незанятую клетку."""
        busy = {coin.cell for coin in self.coins if coin.alive}
        busy.update(player.cell for player in self.players)
        available = [cell for cell in self.free_cells if cell not in busy]
        if not available:
            return
        cell_x, cell_y = random.choice(available)
        self.coins.append(Coin(cell_x, cell_y, gold=random.random() < s.GOLD_CHANCE))

    def apply_input(self, pid: int, data: dict) -> None:
        """Принимает движение игрока."""
        player = self.players[pid]
        player.input = {"dx": float(data.get("dx", 0.0)),
                        "dy": float(data.get("dy", 0.0))}

    def update(self, dt: float) -> None:
        """Продвигает шахту на dt секунд."""
        if self.over:
            return
        self.time_left = max(0.0, self.time_left - dt)
        for player in self.players:
            self._update_player(player, dt)
        for coin in self.coins:
            coin.update(dt)
        self._collect_coins()
        self.coins = [coin for coin in self.coins if coin.alive]
        self._spawn_tick(dt)
        if self.time_left <= 0:
            self._finish()

    def _update_player(self, player: Player, dt: float) -> None:
        """Двигает старателя со скольжением вдоль стен."""
        speed = player.speed
        player.boost = max(0.0, player.boost - dt)
        dx, dy = player.input["dx"], player.input["dy"]
        length = (dx * dx + dy * dy) ** 0.5
        if length == 0:
            return
        dx, dy = dx / length, dy / length
        player.dir_x, player.dir_y = dx, dy
        step = speed * dt
        if not self.hits_wall(player.x + dx * step, player.y, player.radius):
            player.x += dx * step
        if not self.hits_wall(player.x, player.y + dy * step, player.radius):
            player.y += dy * step

    def _collect_coins(self) -> None:
        """Отдаёт каждую монету только одному игроку."""
        for coin in self.coins:
            if not coin.alive:
                continue
            for player in self.players:
                if player.overlaps(coin):
                    coin.alive = False
                    player.take_coin(coin.value)
                    break

    def _spawn_tick(self, dt: float) -> None:
        """Появляет монеты по таймеру."""
        self.spawn_timer -= dt
        while self.spawn_timer <= 0:
            self.spawn_timer += s.SPAWN_INTERVAL
            if len(self.coins) < s.COINS_ON_FIELD:
                self.spawn_coin()

    def _finish(self) -> None:
        """Определяет победителя по очкам."""
        self.over = True
        best = max(player.score for player in self.players)
        leaders = [player for player in self.players if player.score == best]
        self.winner_id = leaders[0].pid if len(leaders) == 1 else None

    def state(self) -> dict:
        """Снимок шахты для консоли или сети."""
        return {"time_left": round(self.time_left, 2),
                "players": [player.to_dict() for player in self.players],
                "coins": [coin.to_dict() for coin in self.coins],
                "over": self.over, "winner": self.winner_id}

# === storage.py ===
"""Таблица рекордов Золотой лихорадки."""
import json
import os
from datetime import datetime
import settings as s

RECORDS_FILE = "records.json"


def load_records(path: str = RECORDS_FILE) -> list[dict]:
    """Читает таблицу или начинает пустую при ошибке файла."""
    if not os.path.exists(path):
        return []
    try:
        with open(path, "r", encoding="utf-8") as file:
            data = json.load(file)
    except (json.JSONDecodeError, UnicodeDecodeError):
        print(f"Файл {path} повреждён, начинаем таблицу заново.")
        return []
    except OSError as error:
        print(f"Не удалось прочитать {path}: {error}")
        return []
    if not isinstance(data, list):
        return []
    return [record for record in data if isinstance(record, dict)]


def save_records(records: list[dict], path: str = RECORDS_FILE) -> None:
    """Сохраняет таблицу в UTF-8."""
    try:
        with open(path, "w", encoding="utf-8") as file:
            json.dump(records, file, ensure_ascii=False, indent=2)
    except OSError as error:
        print(f"Не удалось сохранить {path}: {error}")


def save_match(winner: str, score: str, best: int, path: str = RECORDS_FILE) -> None:
    """Добавляет итог матча."""
    records = load_records(path)
    records.append({"winner": winner, "score": score, "best": best,
                    "date": datetime.now().strftime("%d.%m.%Y %H:%M")})
    save_records(records, path)


def print_table(path: str = RECORDS_FILE) -> None:
    """Печатает лучшие результаты."""
    print("\n--- Таблица рекордов ---")
    records = sorted(load_records(path), key=lambda record: record.get("best", 0), reverse=True)
    for number, record in enumerate(records[:s.TABLE_SIZE], start=1):
        print(f'{number}. {record.get("winner", "?")}  '
              f'{record.get("score", "?")}  '
              f'{record.get("best", 0)}  {record.get("date", "?")}')

# === main.py ===
"""Консольная оболочка Золотой лихорадки."""
import settings as s
from game import Game
import storage

TURN_DT = s.CELL / s.PLAYER_SPEED
KEYS = {"w": (0, -1), "s": (0, 1), "a": (-1, 0), "d": (1, 0)}
SYMBOLS = ["A", "B"]


def draw(state: dict) -> None:
    """Печатает поле по снимку ядра."""
    picture = [list(row) for row in s.MAP_ROWS]
    for coin in state["coins"]:
        x = int(coin["x"] // s.CELL)
        y = int(coin["y"] // s.CELL)
        picture[y][x] = "G" if coin["gold"] else "o"
    for player in state["players"]:
        x = int(player["x"] // s.CELL)
        y = int(player["y"] // s.CELL)
        picture[y][x] = SYMBOLS[player["id"]]
    print()
    for row in picture:
        print(" ".join(row))
    print(f'Время: {state["time_left"]:.1f} с')
    for player in state["players"]:
        print(f'{player["name"]}: {player["score"]} очков, монет: {player["coins"]}')


def parse_command(command: str) -> tuple[str, int]:
    """Разбирает движение и число повторений."""
    command = command.strip().lower()
    if not command:
        return "", 1
    if command[0] in KEYS:
        key = command[0]
        count_text = command[1:]
    else:
        key = ""
        count_text = command
    if count_text and not count_text.isdigit():
        return "?", 0
    try:
        count = int(count_text) if count_text else 1
    except ValueError:
        return "?", 0
    return key, count


def main() -> None:
    """Играет матч и записывает результат."""
    game = Game()
    turns = 0
    while not game.over:
        draw(game.state())
        current = game.players[turns % 2]
        command = input(f"Ход {current.name} (w/a/s/d, d4, 8, q): ").strip().lower()
        if command == "q":
            print("Матч прерван.")
            return
        key, repeat = parse_command(command)
        if key == "?" or repeat <= 0:
            print("Введите направление и число повторов либо число ожидания.")
            continue
        for _ in range(repeat):
            if game.over:
                break
            dx, dy = KEYS.get(key, (0, 0))
            game.apply_input(current.pid, {"dx": dx, "dy": dy})
            game.update(TURN_DT)
            game.apply_input(current.pid, {})
        turns += 1
    draw(game.state())
    score = f"{game.players[0].score}:{game.players[1].score}"
    winner = "Ничья" if game.winner_id is None else game.players[game.winner_id].name
    print(f"Матч окончен. {winner}. Счёт {score}.")
    storage.save_match(winner, score, max(player.score for player in game.players))
    storage.print_table()


if __name__ == "__main__":
    main()
